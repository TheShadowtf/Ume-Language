Standard Library

System & FileSystem

Process and environment utilities via System, and filesystem operations via FileSystem.

Two ways to import
import std.system.*;   // AST-level (recommended)
#include <system.ume>   // textual include (C-style)

Note: System is also always available natively without any import.

System

A static class for interacting with the host process and operating system: environment variables, exit codes, command-line arguments, current time, shell execution, and platform detection.

MethodSignatureDescription
getEnvpublic static func string? getEnv(string name)Environment variable value, or null if unset.
exitpublic static func void exit(int code)Terminate the process with an exit code.
currentTimeMillispublic static func long currentTimeMillis()Epoch milliseconds — useful for timing and seeding RNGs.
argspublic static func List<string> args()Command-line arguments (excluding the program name).
execpublic static func int exec(string command)Run a shell command; return its exit code.
cwdpublic static func string cwd()Current working directory.
platformpublic static func string platform()"windows", "linux", or "macos".
 
import std.system.*;

func void main() {
    // Environment
    string? home = System.getEnv("HOME");
    if (home != null) {
        Console.println("HOME = " + home);
    }

    // Platform
    Console.println("Running on " + System.platform());

    // Current working directory
    Console.println("CWD: " + System.cwd());

    // Command-line arguments
    List<string> args = System.args();
    foreach (string a in args) {
        Console.println("arg: " + a);
    }

    // Timing
    long start = System.currentTimeMillis();
    doExpensiveWork();
    long elapsed = System.currentTimeMillis() - start;
    Console.println($"took {elapsed} ms");

    // Shell execution (returns exit code)
    int rc = System.exec("echo hello");
    Console.println("exit code: " + rc);
}

FileSystem

A static class for filesystem operations: existence checks, directory creation, file/directory listing, copying, moving, and path manipulation. These are all static methods — call them on the class name.

MethodSignatureDescription
existspublic static func bool exists(string path)True if file or directory exists.
isFilepublic static func bool isFile(string path)True if path is a regular file.
isDirectorypublic static func bool isDirectory(string path)True if path is a directory.
createDirpublic static func void createDir(string path)Create one directory.
createDirspublic static func void createDirs(string path)Create all missing directories in a path.
deleteFilepublic static func void deleteFile(string path)Delete a file.
deleteDirpublic static func void deleteDir(string path)Delete a directory.
copypublic static func void copy(string from, string to)Copy a file.
movepublic static func void move(string from, string to)Move/rename a file.
readTextpublic static func string readText(string path)Read entire file as a string.
listDirpublic static func List<string> listDir(string path)Names of entries in a directory.
fileSizepublic static func long fileSize(string path)Size in bytes.
extensionpublic static func string extension(string path)File extension (e.g. ".txt").
filenamepublic static func string filename(string path)Just the filename portion.
parentDirpublic static func string parentDir(string path)The directory containing the path.
joinpublic static func string join(string base, string part)Join two path components with the OS separator.
 
import std.system.*;
import std.collections.*;

func void main() {
    // Create a directory tree
    string dir = "output/data";
    if (!FileSystem.exists(dir)) {
        FileSystem.createDirs(dir);
    }

    // Write a file
    FileSystem.copy("source.txt", FileSystem.join(dir, "copy.txt"));

    // List directory contents
    List<string> entries = FileSystem.listDir(dir);
    foreach (string entry in entries) {
        string full = FileSystem.join(dir, entry);
        if (FileSystem.isFile(full)) {
            long size = FileSystem.fileSize(full);
            Console.println($"{entry} ({size} bytes)");
        } else {
            Console.println(entry + "/");
        }
    }

    // Path manipulation
    string path = "src/main.ume";
    Console.println(FileSystem.filename(path));    // main.ume
    Console.println(FileSystem.extension(path));   // .ume
    Console.println(FileSystem.parentDir(path));   // src
