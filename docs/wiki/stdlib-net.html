Standard Library

Networking

HTTP and TCP clients for network operations. HttpClient for high-level request/response, TcpSocket/TcpListener for raw TCP, and Url for URL encoding and query-string parsing.

Two ways to import
import std.net.*;   // AST-level (recommended)
#include <net.ume>   // textual include (C-style)

HttpResponse

A simple struct holding the result of an HTTP request. The statusCode is the HTTP status (200, 404, etc.); body is the response body as a string; headers is a map of response headers.

FieldType
statusCodepublic int
bodypublic string
headerspublic Map<string, string>

HttpClient

A high-level HTTP client supporting GET and POST requests, custom headers, and timeouts. Construct an instance, optionally configure default headers and timeout, then call get or post.

MethodSignatureDescription
constructorpublic func void constructor()Default settings.
getpublic func HttpResponse get(string url)GET request.
getpublic func HttpResponse get(string url, Map<string,string> headers)GET with custom headers.
postpublic func HttpResponse post(string url, string body)POST with body.
postJsonpublic func HttpResponse postJson(string url, string jsonBody)POST with Content-Type: application/json.
setHeaderpublic func void setHeader(string name, string value)Default header for all subsequent requests.
setTimeoutpublic func void setTimeout(int milliseconds)Per-request timeout.
 
import std.net.*;
import std.collections.*;

func void main() {
    var client = new HttpClient();
    client.setTimeout(5000);

    // Simple GET
    HttpResponse resp = client.get("https://httpbin.org/get");
    Console.println("Status: " + resp.statusCode);
    Console.println("Body: " + resp.body);

    // GET with custom headers
    Map<string, string> headers = new Map<string, string>();
    headers.put("Authorization", "Bearer mytoken");
    HttpResponse authResp = client.get("https://api.example.com/me", headers);

    // POST JSON
    string json = "{\"name\":\"Alice\",\"age\":30}";
    HttpResponse postResp = client.postJson("https://api.example.com/users", json);
    Console.println("Created: " + postResp.statusCode);
}

TcpSocket

A raw TCP client. Connect to a host and port, send and receive strings, then close. Use this for protocols that aren't HTTP, or when you need fine-grained control over the connection.

MethodSignatureDescription
constructorpublic func void constructor()Unconnected socket.
connectpublic func void connect(string host, int port)Establish a connection.
closepublic func void close()Close the connection.
sendpublic func void send(string data)Send a string.
receivepublic func string receive(int maxBytes)Read up to maxBytes.
isConnectedpublic func bool isConnected()Connection state.
 
import std.net.*;

func void main() {
    var sock = new TcpSocket();
    sock.connect("example.com", 80);

    sock.send("GET / HTTP/1.1\r\nHost: example.com\r\nConnection: close\r\n\r\n");

    string response = "";
    while (sock.isConnected()) {
        string chunk = sock.receive(4096);
        if (chunk.length() == 0) break;
        response = response + chunk;
    }

    Console.println(response);
    sock.close();
}

TcpListener

A TCP server. Bind to a port, call accept() to wait for incoming connections, and get a TcpSocket for each client. Typical use: a loop that accepts connections and hands them off to worker threads.

MethodSignature
constructorpublic func void constructor(int port)
startpublic func void start() — begin listening
acceptpublic func TcpSocket accept() — blocks until a client connects
closepublic func void close()
 
import std.net.*;
import std.thread.*;

func void main() {
    var server = new TcpListener(8080);
    server.start();
    Console.println("Listening on :8080");

    while (true) {
        TcpSocket client = server.accept();
        Console.println("Client connected");

        // Handle in a worker thread
        var handler = () -> {
            string req = client.receive(4096);
            client.send("HTTP/1.1 200 OK\r\nContent-Length: 12\r\n\r\nHello, client");
            client.close();
        };
        new Thread(handler).start();
    }
}

Url

Static utilities for URL encoding, decoding, and parsing query strings.

MethodSignatureDescription
encodepublic static func string encode(string raw)URL-encode (percent-encode) a string.
decodepublic static func string decode(string encoded)URL-decode a percent-encoded string.
parseQuerypublic static func Map<string, string> parseQuery(string queryString)Parse key=value&key=value into a map.
 
import std.net.*;
import std.collections.*;

string encoded = Url.encode("hello world & friends");
Console.println(encoded);   // hello+world+%26+friends

string decoded = Url.decode(encoded);
Console.println(decoded);   // hello world & friends

Map<string, string> params = Url.parseQuery("name=Alice&age=30&city=NYC");
Console.println(params.get("name"));   // Alice
Console.println(params.get("age"));    // 30
